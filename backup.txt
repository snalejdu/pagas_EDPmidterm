    <script>
      
        (function(){
            const layers = Array.from(document.querySelectorAll('.parallax-layer'));
            if(!layers.length) return;

            let latestKnownScrollY = 0;
            let ticking = false;

            function onScroll(){
                latestKnownScrollY = window.scrollY || window.pageYOffset;
                requestTick();
            }

            function requestTick(){
                if(!ticking){
                    requestAnimationFrame(update);
                }
                ticking = true;
            }

            function update(){
                const sc = latestKnownScrollY;
                layers.forEach(el => {
                    const depth = parseFloat(el.dataset.depth) || 0;
                    const movement = Math.round(sc * depth);
                    el.style.transform = `translate3d(0, ${movement}px, 0)`;
                });
                ticking = false;
            }

            // Initialize positions (in case page isn't at top)
            onScroll();

            window.addEventListener('scroll', onScroll, {passive:true});
            window.addEventListener('resize', requestTick);
        })();
    </script>
